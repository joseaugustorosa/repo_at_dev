### SEM response_model dedicado (response_model = a própria tabela, como no Starter Kit) — versão vulnerável

{

  "id": 1,
  "reason": "Retorno de exames",
  "notes": "Paciente refere melhora. PA 120/80.",
  "internal_audit_note": "revisar-depois",
  "patient_id": 1,
  "professional_id": 3,
  "status": "agendada",
  "created_from_ip": "10.0.0.5"
}
